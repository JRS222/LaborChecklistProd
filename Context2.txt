$base = "https://webeor.usps.gov"
$session = New-Object Microsoft.PowerShell.Commands.WebRequestSession

$landing = Invoke-WebRequest -Uri "$base/GUI_Login_View.asp" `
        -WebSession $session -UseDefaultCredentials -UseBasicParsing
"Landing: $($landing.BaseResponse.ResponseUri.AbsoluteUri)  Bytes: $($landing.RawContentLength)"

# ---- Dump every form; note which ones contain a <select> ----
"`n=== FORMS ==="
$forms = [regex]::Matches($landing.Content, '<form\b[\s\S]*?</form>', 'IgnoreCase')
"Form count: $($forms.Count)"
$targetForm = $null
foreach ($fm in $forms) {
    $f = $fm.Value
    $action = [regex]::Match($f, 'action\s*=\s*["'']([^"'']+)', 'IgnoreCase').Groups[1].Value
    $method = [regex]::Match($f, 'method\s*=\s*["'']([^"'']+)', 'IgnoreCase').Groups[1].Value
    $name   = [regex]::Match($f, 'name\s*=\s*["'']([^"'']+)',   'IgnoreCase').Groups[1].Value
    $hasSel = $f -match '<select'
    "FORM name='$name' method='$method' action='$action' hasSelect=$hasSel"
    if ($hasSel -and -not $targetForm) { $targetForm = $f }
}

if ($targetForm) {
    "`n=== TARGET FORM FIELDS ==="
    [regex]::Matches($targetForm, '<input\b[^>]*>', 'IgnoreCase') | ForEach-Object { "  INPUT : $($_.Value)" }
    [regex]::Matches($targetForm, '<select\b[^>]*>', 'IgnoreCase') | ForEach-Object { "  SELECT: $($_.Value)" }
    # Show any JS submit/onchange handlers
    [regex]::Matches($targetForm, '(onclick|onchange|onsubmit)\s*=\s*["''][^"'']+["'']', 'IgnoreCase') |
        ForEach-Object { "  HANDLER: $($_.Value)" }
}

# ---- Try bypassing the form: does the viewer accept a site_id in the query string? ----
"`n=== DIRECT VIEWER URL ATTEMPTS (site 76003 = ABILENE TX P&DC) ==="
foreach ($p in @("site_id","SiteID","site","SiteId","loc","location")) {
    try {
        $r = Invoke-WebRequest -Uri "$base/EORViewer/GUI_EORView_Viewer.asp?$p=76003" `
                -WebSession $session -UseDefaultCredentials -UseBasicParsing -ErrorAction Stop
        $u = $r.BaseResponse.ResponseUri.AbsoluteUri
        $ok = $u -notmatch 'GUI_Login_View\.asp'
        "  ?$p=76003 => $(if($ok){'*** VIEWER LOADED ***'}else{'bounced'})  ($($r.RawContentLength) bytes)  $u"
    } catch { "  ?$p=76003 => ERROR: $($_.Exception.Message)" }
}
