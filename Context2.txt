function Connect-WebEor {
    param([int]$SiteId)
    $s = New-Object Microsoft.PowerShell.Commands.WebRequestSession
    $s.UserAgent = $script:UA
    $null = Invoke-WebRequest -Uri "$($script:Base)/GUI_Login_View.asp" `
        -WebSession $s -UseDefaultCredentials -UseBasicParsing
    $body = @{
        User="View"; Pword="View"; ChildSites=""; ServerType="Production"
        GroupList=""; FullName=""; EMail=""; ParentSiteID="$SiteId"; btnLogin="View"
    }
    $r = Invoke-WebRequest -Uri "$($script:Base)/GUI_Login_Validate.asp" -Method POST `
        -Body $body -WebSession $s -UseDefaultCredentials -UseBasicParsing `
        -Headers @{ Origin = $script:Base; Referer = "$($script:Base)/GUI_Login_View.asp" }
    if ($r.Content -notmatch 'DMS/EOR Main Menu') {
        throw "Login failed for site $SiteId (got $($r.RawContentLength) bytes)"
    }
    return $s
}

function Get-SiteList {
    $s = New-Object Microsoft.PowerShell.Commands.WebRequestSession
    $s.UserAgent = $script:UA
    $landing = Invoke-WebRequest -Uri "$($script:Base)/GUI_Login_View.asp" `
        -WebSession $s -UseDefaultCredentials -UseBasicParsing
    [regex]::Matches($landing.Content,
        '<option\s+value\s*="(?<id>\d+)"\s*>(?<name>[^<]+)</option>', 'IgnoreCase') |
        ForEach-Object {
            [pscustomobject]@{
                Id   = $_.Groups['id'].Value
                Name = [System.Net.WebUtility]::HtmlDecode($_.Groups['name'].Value.Trim())
            }
        } | Sort-Object Name
}

function Get-EorSql {
    param(
        [Parameter(Mandatory)] [int]      $SiteId,
        [Parameter(Mandatory)] [datetime] $From,
        [Parameter(Mandatory)] [datetime] $To,
        [Parameter(Mandatory)] [bool]     $DateOnly
    )
    $ci = [System.Globalization.CultureInfo]::InvariantCulture

    if ($DateOnly) {
        $df = $From.ToString("dd-MMM-yyyy", $ci)
        $dt = $To  .ToString("dd-MMM-yyyy", $ci)
        $where = "application_data.mods_date >= TO_DATE('$df', 'DD-MON-YYYY') AND application_data.mods_date <  TO_DATE('$dt', 'DD-MON-YYYY') + 1"
    } else {
        $ff = $From.ToString("dd-MMM-yyyy HH:mm:ss", $ci)
        $tt = $To  .ToString("dd-MMM-yyyy HH:mm:ss", $ci)
        $where = "application_data.mods_date >= TO_DATE('$ff', 'DD-MON-YYYY HH24:MI:SS') AND application_data.mods_date <= TO_DATE('$tt', 'DD-MON-YYYY HH24:MI:SS')"
    }

    @"
SELECT /*+ ordered */ SUBSTR( LPAD( postal_facility.zip_code_id, 9, '0' ), 1, 5 ) || '-' || SUBSTR( LPAD( postal_facility.zip_code_id, 9, '0' ), 6 ) site_code, machine_type.mach_type_desc, machine.machine_nbr, TRIM(application_data.machine_sort_program_name) msp, application_data.tour_code, TO_CHAR( application_data.start_dtm, 'MM/DD/YY HH24:MI') start_date, TO_CHAR( application_data.end_dtm, 'MM/DD/YY HH24:MI') end_date, GUI_Common_Helper.FormatOpNbr( application_data.mail_operation_nbr ) op_nbr, application_data.app_sequence_nbr, application_data.total_pieces_fed_cnt, GUI_Common_Helper.GetRunTransStatus( application_data.mail_operation_nbr, application_data.mods_trans_ind, application_data.mods_trans_date ) ModsTrans, DECODE( UPPER( application_data.dsis_ind ), 'T', 'S', 'NS' ) dsis, end_of_run.run_nbr, application_data.run_sequence_nbr , substr(TO_CHAR(application_data.mods_date, 'MM/DD/YY HH24:MI'), 7, 2) mYear, substr(TO_CHAR(application_data.start_dtm, 'MM/DD/YY HH24:MI'), 7, 2) sYear, substr(TO_CHAR(application_data.end_dtm, 'MM/DD/YY HH24:MI'), 7, 2) eYear, NVL(substr(TO_CHAR(application_data.mods_trans_date), 9, 2), '00') mTransYear, postal_facility.site_name FROM EOR_PSBRPA.application_data, EOR_PSBRPA.end_of_run, ( SELECT site_id, site_name, zip_code_id FROM EOR_PSBRPA.postal_facility x START WITH x.site_id = $SiteId CONNECT BY PRIOR x.site_id = x.parent_site_id AND x.site_id <> $SiteId ) postal_facility, EOR_PSBRPA.machine_type, EOR_PSBRPA.machine WHERE end_of_run.site_id = postal_facility.site_id AND end_of_run.machine_id = machine.machine_id AND end_of_run.site_id = machine.site_id AND end_of_run.run_mach_type_code = machine_type.mach_type_code AND end_of_run.run_sequence_nbr = application_data.run_sequence_nbr AND $where ORDER BY mach_type_desc asc, machine_nbr asc,  myear asc, syear asc, start_date asc
"@
}

function Save-EorCsv {
    param(
        [Parameter(Mandatory)] $Session,
        [Parameter(Mandatory)] [int]      $SiteId,
        [Parameter(Mandatory)] [datetime] $From,
        [Parameter(Mandatory)] [datetime] $To,
        [Parameter(Mandatory)] [bool]     $DateOnly,
        [Parameter(Mandatory)] [string]   $OutFile
    )
    $body = @{
        tabCsvGroup   = " "
        tabCsvHeader  = "Site,MType,MNo,Op No.,Sort Program,Tour,Run#,Start,End,Fed,MODS,DOIS"
        tabCsvColumns = "0,1,2,7,3,4,12,5,6,9,10,11"
        tabCsvSource  = "CLIENT"
        tabCsvSQL     = Get-EorSql -SiteId $SiteId -From $From -To $To -DateOnly $DateOnly
    }
    $r = Invoke-WebRequest -Uri "$($script:Base)/ReportPage/GUI_Report_CSV.asp" `
        -Method POST -Body $body -WebSession $Session -UseDefaultCredentials -UseBasicParsing `
        -Headers @{ Origin = $script:Base; Referer = "$($script:Base)/EORViewer/GUI_EORView_Viewer.asp" } `
        -OutFile $OutFile
    if ($r.Headers['Content-Type'] -notmatch 'csv|excel') {
        throw "CSV failed: Content-Type=$($r.Headers['Content-Type']) size=$($r.RawContentLength)"
    }
    return (Get-Item $OutFile)
}

# =====================================================================

function Write-Log([string]$msg) {
    $ts = Get-Date -Format "HH:mm:ss"
    $txtLog.AppendText("[$ts] $msg`r`n")
    $txtLog.SelectionStart = $txtLog.Text.Length
    $txtLog.ScrollToCaret()
    [System.Windows.Forms.Application]::DoEvents()
}

function Set-Status([string]$msg, [string]$color = "Black") {
    $lblStatus.Text = $msg
    $lblStatus.ForeColor = [System.Drawing.Color]::FromName($color)
    [System.Windows.Forms.Application]::DoEvents()
}

function Get-SelectedSite {
    if ($cmbSite.Text -match '^(\d{5})\s+(.+)$') {
        return @{ Id = $Matches[1]; Name = $Matches[2] }
    }
    return $null
}

# Reload sites
$btnReload.Add_Click({
    $btnReload.Enabled = $false
    Set-Status "Loading site list..." "DarkBlue"
    try {
        Write-Log "Fetching site list..."
        $sites = Get-SiteList
        $cmbSite.BeginUpdate()
        $cmbSite.Items.Clear()
        foreach ($s in $sites) { [void]$cmbSite.Items.Add("$($s.Id)  $($s.Name)") }
        $cmbSite.EndUpdate()
        Write-Log "Loaded $($sites.Count) sites"
        Set-Status "Loaded $($sites.Count) sites" "DarkGreen"
    } catch {
        Write-Log "Reload failed: $($_.Exception.Message)"
        Set-Status "Reload failed" "Red"
    } finally {
        $btnReload.Enabled = $true
    }
})

# Date-only checkbox toggles picker format
$chkDateOnly.Add_CheckedChanged({
    if ($chkDateOnly.Checked) {
        $dtpFrom.CustomFormat = 'yyyy-MM-dd'
        $dtpTo.CustomFormat   = 'yyyy-MM-dd'
        $dtpFrom.Value = $dtpFrom.Value.Date
        $dtpTo.Value   = $dtpTo.Value.Date
    } else {
        $dtpFrom.CustomFormat = 'yyyy-MM-dd HH:mm'
        $dtpTo.CustomFormat   = 'yyyy-MM-dd HH:mm'
    }
})

# Browse for output path
$btnBrowse.Add_Click({
    $sfd = New-Object System.Windows.Forms.SaveFileDialog
    $sfd.Filter = "CSV files (*.csv)|*.csv|All files (*.*)|*.*"
    $sfd.InitialDirectory = $script:DownloadsDir
    if ($txtOut.Text) { $sfd.FileName = [System.IO.Path]::GetFileName($txtOut.Text) }
    if ($sfd.ShowDialog() -eq [System.Windows.Forms.DialogResult]::OK) {
        $txtOut.Text = $sfd.FileName
        $script:OutCustom = $true
    }
})

# Download
$btnDownload.Add_Click({
    $site = Get-SelectedSite
    if (-not $site) {
        [System.Windows.Forms.MessageBox]::Show(
            "Pick a site from the dropdown (use the Reload button if the list is empty).",
            "No site", 'OK', 'Warning') | Out-Null
        return
    }

    $dateOnly = $chkDateOnly.Checked
    $from = $dtpFrom.Value
    $to   = $dtpTo.Value

    if ($dateOnly) {
        if ($to.Date -lt $from.Date) {
            [System.Windows.Forms.MessageBox]::Show(
                "'To' date is earlier than 'From' date.", "Bad range", 'OK', 'Warning') | Out-Null
            return
        }
    } elseif ($to -lt $from) {
        [System.Windows.Forms.MessageBox]::Show(
            "'To' is earlier than 'From'.", "Bad range", 'OK', 'Warning') | Out-Null
        return
    }

    # Determine output path
    if ($script:OutCustom -and $txtOut.Text) {
        $outFile = $txtOut.Text
    } else {
        $outFile = Join-Path $script:DownloadsDir `
            "EOR_$($site.Id)_$($from.ToString('yyyyMMdd'))_$($to.ToString('yyyyMMdd')).csv"
        $txtOut.Text = $outFile
    }

    if (Test-Path $outFile) {
        $ans = [System.Windows.Forms.MessageBox]::Show(
            "Overwrite existing file?`r`n$outFile", "File exists", 'YesNo', 'Question')
        if ($ans -ne [System.Windows.Forms.DialogResult]::Yes) { return }
    }

    $btnDownload.Enabled = $false
    $btnReload.Enabled = $false
    try {
        $sid = [int]$site.Id
        if (-not $script:Session -or $script:SessionSite -ne $sid) {
            Set-Status "Logging in as $sid..." "DarkBlue"
            Write-Log "Logging in as $sid  ($($site.Name))"
            $script:Session = Connect-WebEor -SiteId $sid
            $script:SessionSite = $sid
            Write-Log "Login OK"
        }

        Set-Status "Downloading..." "DarkBlue"
        $modeTxt = if ($dateOnly) {
            "date-only $($from.ToString('yyyy-MM-dd'))..$($to.ToString('yyyy-MM-dd'))"
        } else {
            "$($from.ToString('yyyy-MM-dd HH:mm'))..$($to.ToString('yyyy-MM-dd HH:mm'))"
        }
        Write-Log "Downloading CSV  site=$sid  range=$modeTxt"

        $f = Save-EorCsv -Session $script:Session -SiteId $sid `
            -From $from -To $to -DateOnly $dateOnly -OutFile $outFile

        $lines = (Get-Content $f.FullName | Measure-Object -Line).Lines
        Write-Log "Saved: $($f.FullName)  ($([math]::Round($f.Length/1KB,1)) KB, ~$($lines - 1) rows)"
        Set-Status "Done: $($f.Name)  ($($lines - 1) rows)" "DarkGreen"
    } catch {
        Write-Log "FAILED: $($_.Exception.Message)"
        Set-Status "Failed" "Red"
        # Invalidate session so the next click logs in again
        $script:Session = $null
        $script:SessionSite = $null
    } finally {
        $btnDownload.Enabled = $true
        $btnReload.Enabled = $true
    }
})
