$backend = "https://eagnmnwbp13ec.usps.gov"

# 1. Get the full frameset HTML (previous run truncated it at 300 chars)
$fs = Invoke-WebRequest -Uri "$backend/Common/GUI_Common_framemenus.asp" `
        -WebSession $session -UseDefaultCredentials -UseBasicParsing
"=== Full frameset HTML ==="
$fs.Content
"==========================="

# 2. Extract every FRAME SRC
$frameSrcs = [regex]::Matches($fs.Content,
    '<FRAME[^>]*\bSRC\s*=\s*["'']([^"'']+)["'']', 'IgnoreCase') |
    ForEach-Object { $_.Groups[1].Value }
"Frames found: $($frameSrcs.Count) -> $($frameSrcs -join ', ')"

# 3. Fetch each frame in order (this mirrors the browser)
$baseUri = [uri]"$backend/Common/GUI_Common_framemenus.asp"
foreach ($src in $frameSrcs) {
    $url = [uri]::new($baseUri, $src).AbsoluteUri
    try {
        $fr = Invoke-WebRequest -Uri $url -WebSession $session `
                -UseDefaultCredentials -UseBasicParsing
        "  Frame: $url"
        "     Status=$($fr.StatusCode)  Bytes=$($fr.RawContentLength)  FinalUri=$($fr.BaseResponse.ResponseUri.AbsoluteUri)"
        "     Body head: $($fr.Content.Substring(0,[Math]::Min(200,$fr.Content.Length)))"
    } catch {
        "  Frame: $url  ERROR: $($_.Exception.Message)"
    }
}

# 4. Retry viewer on the backend — did loading the frames help?
$view2 = Invoke-WebRequest -Uri "$backend/EORViewer/GUI_EORView_Viewer.asp" `
        -WebSession $session -UseDefaultCredentials -UseBasicParsing
"`n=== Viewer after frames ==="
"  FinalUri : $($view2.BaseResponse.ResponseUri.AbsoluteUri)"
"  Bytes    : $($view2.RawContentLength)"
"  Loaded   : $($view2.BaseResponse.ResponseUri.AbsoluteUri -match 'EORViewer')"

# 5. Shortcut check — does the CSV POST work NOW, without the viewer?
#    (Tests whether auth is session-state-only or actually viewer-page-bound.)
$sql = @'
SELECT /*+ ordered */ SUBSTR( LPAD( postal_facility.zip_code_id, 9, '0' ), 1, 5 ) || '-' || SUBSTR( LPAD( postal_facility.zip_code_id, 9, '0' ), 6 ) site_code, machine_type.mach_type_desc, machine.machine_nbr, TRIM(application_data.machine_sort_program_name) msp, application_data.tour_code, TO_CHAR( application_data.start_dtm, 'MM/DD/YY HH24:MI') start_date, TO_CHAR( application_data.end_dtm, 'MM/DD/YY HH24:MI') end_date, GUI_Common_Helper.FormatOpNbr( application_data.mail_operation_nbr ) op_nbr, application_data.app_sequence_nbr, application_data.total_pieces_fed_cnt, GUI_Common_Helper.GetRunTransStatus( application_data.mail_operation_nbr, application_data.mods_trans_ind, application_data.mods_trans_date ) ModsTrans, DECODE( UPPER( application_data.dsis_ind ), 'T', 'S', 'NS' ) dsis, end_of_run.run_nbr, application_data.run_sequence_nbr , substr(TO_CHAR(application_data.mods_date, 'MM/DD/YY HH24:MI'), 7, 2) mYear, substr(TO_CHAR(application_data.start_dtm, 'MM/DD/YY HH24:MI'), 7, 2) sYear, substr(TO_CHAR(application_data.end_dtm, 'MM/DD/YY HH24:MI'), 7, 2) eYear, NVL(substr(TO_CHAR(application_data.mods_trans_date), 9, 2), '00') mTransYear, postal_facility.site_name FROM EOR_PSBRPA.application_data, EOR_PSBRPA.end_of_run, ( SELECT site_id, site_name, zip_code_id FROM EOR_PSBRPA.postal_facility x START WITH x.site_id = 76003 CONNECT BY PRIOR x.site_id = x.parent_site_id AND x.site_id <> 76003 ) postal_facility, EOR_PSBRPA.machine_type, EOR_PSBRPA.machine WHERE end_of_run.site_id = postal_facility.site_id AND end_of_run.machine_id = machine.machine_id AND end_of_run.site_id = machine.site_id AND end_of_run.run_mach_type_code = machine_type.mach_type_code AND end_of_run.run_sequence_nbr = application_data.run_sequence_nbr AND application_data.mods_date >= TO_DATE('17-Oct-2026', 'DD-MON-YYYY') AND application_data.mods_date <= TO_DATE('17-Oct-2026', 'DD-MON-YYYY') ORDER BY mach_type_desc asc, machine_nbr asc,  myear asc, syear asc, start_date asc
'@
$csvBody = @{
    tabCsvGroup   = " "
    tabCsvHeader  = "Site,MType,MNo,Op No.,Sort Program,Tour,Run#,Start,End,Fed,MODS,DOIS"
    tabCsvColumns = "0,1,2,7,3,4,12,5,6,9,10,11"
    tabCsvSource  = "CLIENT"
    tabCsvSQL     = $sql
}
$csv2 = Invoke-WebRequest -Uri "$backend/ReportPage/GUI_Report_CSV.asp" `
        -Method POST -Body $csvBody -WebSession $session `
        -UseDefaultCredentials -UseBasicParsing
"`n=== CSV POST (backend, pre-viewer) ==="
"  Status : $($csv2.StatusCode)"
"  CT     : $($csv2.Headers['Content-Type'])"
"  Bytes  : $($csv2.RawContentLength)"
"  First 300:"
$csv2.Content.Substring(0, [Math]::Min(300, $csv2.Content.Length))
