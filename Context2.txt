$base = "https://webeor.usps.gov"
$session = New-Object Microsoft.PowerShell.Commands.WebRequestSession

# 1. Get the landing page (sets initial cookies)
$null = Invoke-WebRequest -Uri "$base/GUI_Login_View.asp" `
        -WebSession $session -UseDefaultCredentials -UseBasicParsing

# 2. Submit the site-selection form (View mode, site 76003 = ABILENE TX P&DC)
$body = @{
    User         = "View"
    Pword        = "View"
    ChildSites   = ""
    ServerType   = "Production"
    GroupList    = ""
    FullName     = ""
    EMail        = ""
    ParentSiteID = "76003"
    btnLogin     = "View"
}

$post = Invoke-WebRequest -Uri "$base/GUI_Login_Validate.asp" `
        -Method POST -Body $body -WebSession $session `
        -UseDefaultCredentials -UseBasicParsing

"POST Status:   $($post.StatusCode)"
"POST FinalUri: $($post.BaseResponse.ResponseUri.AbsoluteUri)"
"POST Bytes:    $($post.RawContentLength)"
"POST CT:       $($post.Headers['Content-Type'])"

# Show JS redirect if server used one instead of 302
$js = [regex]::Match($post.Content,
      '(?:window\.|top\.|self\.)?location(?:\.href)?\s*=\s*["'']([^"'']+)', 'IgnoreCase')
"JS redirect:   $(if($js.Success){$js.Groups[1].Value}else{'(none)'})"

# 3. Does the viewer accept us now?
$view = Invoke-WebRequest -Uri "$base/EORViewer/GUI_EORView_Viewer.asp" `
        -WebSession $session -UseDefaultCredentials -UseBasicParsing
"Viewer FinalUri: $($view.BaseResponse.ResponseUri.AbsoluteUri)"
"Viewer Bytes:    $($view.RawContentLength)"
"Viewer loaded:   $($view.BaseResponse.ResponseUri.AbsoluteUri -match 'EORViewer')"

# 4. Show what cookies the session now has
"`n=== Cookies ==="
$session.Cookies.GetCookies($base) | ForEach-Object { "  $($_.Name) = $($_.Value)" }
