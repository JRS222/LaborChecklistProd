$front = "https://webeor.usps.gov"
$session = New-Object Microsoft.PowerShell.Commands.WebRequestSession
$session.UserAgent = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/155.0.0.0 Safari/537.36 Edg/155.0.0.0"

function ShowCookies([string]$label) {
    $asps = $session.Cookies.GetCookies($front) |
            Where-Object { $_.Name -like "ASPSESSIONID*" } |
            ForEach-Object { "$($_.Name)=$($_.Value)" }
    "$label : $asps"
}

# 1. Landing
$null = Invoke-WebRequest -Uri "$front/GUI_Login_View.asp" -WebSession $session `
    -UseDefaultCredentials -UseBasicParsing
ShowCookies "after landing"

# 2. Site-select POST (with full browser headers)
$loginBody = @{
    User="View"; Pword="View"; ChildSites=""; ServerType="Production"
    GroupList=""; FullName=""; EMail=""; ParentSiteID="19010"; btnLogin="View"
}
$post1 = Invoke-WebRequest -Uri "$front/GUI_Login_Validate.asp" -Method POST -Body $loginBody `
    -WebSession $session -UseDefaultCredentials -UseBasicParsing `
    -Headers @{
        Origin  = $front
        Referer = "$front/GUI_Login_View.asp"
    }
ShowCookies "after Validate POST"
"POST Status=$($post1.StatusCode)  Bytes=$($post1.RawContentLength)"
"POST body head: $($post1.Content.Substring(0,[Math]::Min(400,$post1.Content.Length)))"

# 3. Did the response tell us where to go next?
$js = [regex]::Match($post1.Content, 'location(?:\.href)?\s*=\s*["'']([^"'']+)', 'IgnoreCase')
"JS redirect: $(if ($js.Success) {$js.Groups[1].Value} else {'(none)'})"

# 4. Immediate viewer probe
$viewer = Invoke-WebRequest -Uri "$front/EORViewer/GUI_EORView_Viewer.asp" `
    -WebSession $session -UseDefaultCredentials -UseBasicParsing
"Viewer FinalUri: $($viewer.BaseResponse.ResponseUri.AbsoluteUri)"
"Viewer Bytes   : $($viewer.RawContentLength)"
"Viewer authed  : $([bool]($viewer.Content -match 'gstrTableSQL'))"
