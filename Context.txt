$front   = "https://webeor.usps.gov"
$session = New-Object Microsoft.PowerShell.Commands.WebRequestSession

# 1. Landing
$null = Invoke-WebRequest -Uri "$front/GUI_Login_View.asp" -WebSession $session `
        -UseDefaultCredentials -UseBasicParsing

# 2. Login form POST
$loginBody = @{
    User="View"; Pword="View"; ChildSites=""; ServerType="Production"
    GroupList=""; FullName=""; EMail=""; ParentSiteID="19010"; btnLogin="View"
}
$post1 = Invoke-WebRequest -Uri "$front/GUI_Login_Validate.asp" -Method POST -Body $loginBody `
        -WebSession $session -UseDefaultCredentials -UseBasicParsing

# 3. Follow JS redirect to backend
$js1 = [regex]::Match($post1.Content, 'location(?:\.href)?\s*=\s*["'']([^"'']+)', 'IgnoreCase')
$auto = [uri]$js1.Groups[1].Value
"Auto.asp: $auto"
$autoResp = Invoke-WebRequest -Uri $auto -WebSession $session -UseDefaultCredentials -UseBasicParsing
"  Status=$($autoResp.StatusCode)  FinalUri=$($autoResp.BaseResponse.ResponseUri.AbsoluteUri)"

# 4. Frameset (extract frmMain)
$js2 = [regex]::Match($autoResp.Content, 'location(?:\.href)?\s*=\s*["'']([^"'']+)', 'IgnoreCase')
$frameset = [uri]::new($auto, $js2.Groups[1].Value)
$fs = Invoke-WebRequest -Uri $frameset -WebSession $session -UseDefaultCredentials -UseBasicParsing
$frmMainSrc = [regex]::Match($fs.Content, '<FRAME[^>]*\bSRC\s*=\s*["'']([^"'']+GUI_Login_Staged[^"'']*)', 'IgnoreCase').Groups[1].Value
"frmMain  : $frmMainSrc"

# 5. Fetch staged page, POST staged form
$stagedUrl = [uri]::new($frameset, $frmMainSrc)
$staged = Invoke-WebRequest -Uri $stagedUrl -WebSession $session -UseDefaultCredentials -UseBasicParsing
$stagedBody = @{}
[regex]::Matches($staged.Content, '<INPUT[^>]*\bNAME\s*=\s*["'']([^"'']+)["''][^>]*\bVALUE\s*=\s*["'']([^"'']*)["'']', 'IgnoreCase') |
    ForEach-Object { $stagedBody[$_.Groups[1].Value] = $_.Groups[2].Value }

$stagedPost = Invoke-WebRequest -Uri "$front/GUI_Login_Validate.asp" -Method POST `
        -Body $stagedBody -WebSession $session -UseDefaultCredentials -UseBasicParsing
"Staged POST: Status=$($stagedPost.StatusCode)  FinalUri=$($stagedPost.BaseResponse.ResponseUri.AbsoluteUri)"

# 6. ★ THE NEW STEP: viewer on the FRONT-END host
$viewer = Invoke-WebRequest -Uri "$front/EORViewer/GUI_EORView_Viewer.asp" `
        -WebSession $session -UseDefaultCredentials -UseBasicParsing
"Viewer (front): Status=$($viewer.StatusCode)  Bytes=$($viewer.RawContentLength)  FinalUri=$($viewer.BaseResponse.ResponseUri.AbsoluteUri)"
"  Has gstrTableSQL: $([bool]($viewer.Content -match 'gstrTableSQL'))"

# 7. Extract the actual gstrTableSQL from the loaded viewer
$realSql = ""
if ($viewer.Content -match 'gstrTableSQL\s*=\s*"([^"]+)"') {
    $realSql = $Matches[1] -replace '\\x3C', '<' -replace '\\x3E', '>' -replace '&lt;','<' -replace '&gt;','>' -replace '&amp;','&'
    "  SQL matches captured: $($realSql -eq $null -or $realSql.Length -lt 100)"
}

# 8. ★ CSV POST — front-end, using the SQL just extracted from the viewer
if ($realSql) {
    $csvBody = @{
        tabCsvGroup   = " "
        tabCsvHeader  = "Site,MType,MNo,Op No.,Sort Program,Tour,Run#,Start,End,Fed,MODS,DOIS"
        tabCsvColumns = "0,1,2,7,3,4,12,5,6,9,10,11"
        tabCsvSource  = "CLIENT"
        tabCsvSQL     = $realSql
    }
    $csv = Invoke-WebRequest -Uri "$front/ReportPage/GUI_Report_CSV.asp" -Method POST `
            -Body $csvBody -WebSession $session -UseDefaultCredentials -UseBasicParsing
    "`nCSV POST (front): Status=$($csv.StatusCode)  CT=$($csv.Headers['Content-Type'])  Bytes=$($csv.RawContentLength)"
    "  First 400:"
    $csv.Content.Substring(0, [Math]::Min(400, $csv.Content.Length))
} else {
    "  No SQL extracted — viewer didn't load as expected"
}
