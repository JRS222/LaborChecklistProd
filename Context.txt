# =====================================================================
# WebEorDownloader.ps1
#   Downloads the EOR CSV from WebEOR for a given site + date range.
#   Authentication: Windows (Negotiate) via -UseDefaultCredentials.
#   No 7-day limit — the UI enforces it on the viewer, not the CSV.
# =====================================================================

$script:WebEorBase = "https://webeor.usps.gov"
$script:WebEorUA   = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) " +
                     "AppleWebKit/537.36 (KHTML, like Gecko) " +
                     "Chrome/155.0.0.0 Safari/537.36 Edg/155.0.0.0"


# ---------------------------------------------------------------------
# Connect-WebEor
#   Establishes an authenticated session for the given site.
#   Returns @{ Session; Base; SiteId }.
#
#   IMPORTANT: The login POST response contains a JS redirect to
#   /GUI_Login_Main.asp — that is the LOGOUT link. Do not follow it.
#   If the POST returns the main menu HTML, you are authenticated.
# ---------------------------------------------------------------------
function Connect-WebEor {
    param(
        [Parameter(Mandatory)] [int] $SiteId
    )

    $session = New-Object Microsoft.PowerShell.Commands.WebRequestSession
    $session.UserAgent = $script:WebEorUA

    # 1. Landing (establishes ASPSESSIONID + NSC affinity cookie)
    $null = Invoke-WebRequest -Uri "$script:WebEorBase/GUI_Login_View.asp" `
        -WebSession $session -UseDefaultCredentials -UseBasicParsing

    # 2. Site-select POST — Origin + Referer headers are REQUIRED
    $loginBody = @{
        User         = "View"
        Pword        = "View"
        ChildSites   = ""
        ServerType   = "Production"
        GroupList    = ""
        FullName     = ""
        EMail        = ""
        ParentSiteID = "$SiteId"
        btnLogin     = "View"
    }
    $post = Invoke-WebRequest -Uri "$script:WebEorBase/GUI_Login_Validate.asp" `
        -Method POST -Body $loginBody -WebSession $session `
        -UseDefaultCredentials -UseBasicParsing `
        -Headers @{
            Origin  = $script:WebEorBase
            Referer = "$script:WebEorBase/GUI_Login_View.asp"
        }

    # 3. Verify — main menu HTML means authenticated
    if ($post.Content -notmatch 'DMS/EOR Main Menu') {
        $head = if ($post.Content.Length -gt 300) { $post.Content.Substring(0,300) } else { $post.Content }
        throw "WebEOR login failed for site $SiteId " +
              "(got $($post.RawContentLength) bytes, expected main menu). Body: $head"
    }

    return @{
        Session = $session
        Base    = $script:WebEorBase
        SiteId  = [int]$SiteId
    }
}


# ---------------------------------------------------------------------
# Get-SiteList
#   Returns all sites from the landing page as
#   [pscustomobject]@{ Id; Name }, sorted by Name.
# ---------------------------------------------------------------------
function Get-SiteList {
    $session = New-Object Microsoft.PowerShell.Commands.WebRequestSession
    $session.UserAgent = $script:WebEorUA
    $landing = Invoke-WebRequest -Uri "$script:WebEorBase/GUI_Login_View.asp" `
        -WebSession $session -UseDefaultCredentials -UseBasicParsing

    return [regex]::Matches($landing.Content,
        '<option\s+value\s*="(?<id>\d+)"\s*>(?<name>[^<]+)</option>', 'IgnoreCase') |
        ForEach-Object {
            [pscustomobject]@{
                Id   = $_.Groups['id'].Value
                Name = [System.Net.WebUtility]::HtmlDecode($_.Groups['name'].Value.Trim())
            }
        } | Sort-Object Name
}


# ---------------------------------------------------------------------
# Get-SiteMap
#   Returns @{ "SITE NAME" = "ID"; ... } — used when you already
#   have a site name and need the numeric ID (e.g. from Machine
#   Checker.ps1 -Site "PITTSBURGH PA P&DC").
# ---------------------------------------------------------------------
function Get-SiteMap {
    $map = @{}
    Get-SiteList | ForEach-Object { $map[$_.Name] = $_.Id }
    return $map
}


# ---------------------------------------------------------------------
# Get-EorSql
#   Builds the SQL string for the CSV export.
#   Range semantics: inclusive of FromDate..ToDate, using a half-open
#   upper bound (< ToDate + 1) so rows with non-midnight mods_date
#   are included.
# ---------------------------------------------------------------------
function Get-EorSql {
    param(
        [Parameter(Mandatory)] [int]      $SiteId,
        [Parameter(Mandatory)] [datetime] $FromDate,
        [Parameter(Mandatory)] [datetime] $ToDate
    )
    $df = $FromDate.ToString("dd-MMM-yyyy", [System.Globalization.CultureInfo]::InvariantCulture)
    $dt = $ToDate  .ToString("dd-MMM-yyyy", [System.Globalization.CultureInfo]::InvariantCulture)

    return @"
SELECT /*+ ordered */ SUBSTR( LPAD( postal_facility.zip_code_id, 9, '0' ), 1, 5 ) || '-' || SUBSTR( LPAD( postal_facility.zip_code_id, 9, '0' ), 6 ) site_code, machine_type.mach_type_desc, machine.machine_nbr, TRIM(application_data.machine_sort_program_name) msp, application_data.tour_code, TO_CHAR( application_data.start_dtm, 'MM/DD/YY HH24:MI') start_date, TO_CHAR( application_data.end_dtm, 'MM/DD/YY HH24:MI') end_date, GUI_Common_Helper.FormatOpNbr( application_data.mail_operation_nbr ) op_nbr, application_data.app_sequence_nbr, application_data.total_pieces_fed_cnt, GUI_Common_Helper.GetRunTransStatus( application_data.mail_operation_nbr, application_data.mods_trans_ind, application_data.mods_trans_date ) ModsTrans, DECODE( UPPER( application_data.dsis_ind ), 'T', 'S', 'NS' ) dsis, end_of_run.run_nbr, application_data.run_sequence_nbr , substr(TO_CHAR(application_data.mods_date, 'MM/DD/YY HH24:MI'), 7, 2) mYear, substr(TO_CHAR(application_data.start_dtm, 'MM/DD/YY HH24:MI'), 7, 2) sYear, substr(TO_CHAR(application_data.end_dtm, 'MM/DD/YY HH24:MI'), 7, 2) eYear, NVL(substr(TO_CHAR(application_data.mods_trans_date), 9, 2), '00') mTransYear, postal_facility.site_name FROM EOR_PSBRPA.application_data, EOR_PSBRPA.end_of_run, ( SELECT site_id, site_name, zip_code_id FROM EOR_PSBRPA.postal_facility x START WITH x.site_id = $SiteId CONNECT BY PRIOR x.site_id = x.parent_site_id AND x.site_id <> $SiteId ) postal_facility, EOR_PSBRPA.machine_type, EOR_PSBRPA.machine WHERE end_of_run.site_id = postal_facility.site_id AND end_of_run.machine_id = machine.machine_id AND end_of_run.site_id = machine.site_id AND end_of_run.run_mach_type_code = machine_type.mach_type_code AND end_of_run.run_sequence_nbr = application_data.run_sequence_nbr AND application_data.mods_date >= TO_DATE('$df', 'DD-MON-YYYY') AND application_data.mods_date <  TO_DATE('$dt', 'DD-MON-YYYY') + 1 ORDER BY mach_type_desc asc, machine_nbr asc,  myear asc, syear asc, start_date asc
"@
}


# ---------------------------------------------------------------------
# Get-WebEorCsv
#   Downloads the EOR CSV for a site + range. Throws if Content-Type
#   isn't csv/excel. Returns the FileInfo of the saved file.
# ---------------------------------------------------------------------
function Get-WebEorCsv {
    param(
        [Parameter(Mandatory)] $Connection,
        [Parameter(Mandatory)] [int]      $SiteId,
        [Parameter(Mandatory)] [datetime] $FromDate,
        [Parameter(Mandatory)] [datetime] $ToDate,
        [Parameter(Mandatory)] [string]   $OutFile
    )

    $body = @{
        tabCsvGroup   = " "
        tabCsvHeader  = "Site,MType,MNo,Op No.,Sort Program,Tour,Run#,Start,End,Fed,MODS,DOIS"
        tabCsvColumns = "0,1,2,7,3,4,12,5,6,9,10,11"
        tabCsvSource  = "CLIENT"
        tabCsvSQL     = Get-EorSql -SiteId $SiteId -FromDate $FromDate -ToDate $ToDate
    }

    $r = Invoke-WebRequest -Uri "$($Connection.Base)/ReportPage/GUI_Report_CSV.asp" `
        -Method POST -Body $body -WebSession $Connection.Session `
        -UseDefaultCredentials -UseBasicParsing `
        -Headers @{
            Origin  = $Connection.Base
            Referer = "$($Connection.Base)/EORViewer/GUI_EORView_Viewer.asp"
        } `
        -OutFile $OutFile

    $ct = $r.Headers['Content-Type']
    if ($ct -notmatch 'csv|excel') {
        throw "CSV download failed: Content-Type=$ct  size=$($r.RawContentLength)  file=$OutFile"
    }
    return (Get-Item $OutFile)
}


# ---------------------------------------------------------------------
# Select-SiteInteractive
#   Prompts for a partial site name or exact ID. Returns one
#   [pscustomobject]@{Id;Name} or $null on cancel.
#   Type '?' at the prompt to list all sites.
# ---------------------------------------------------------------------
function Select-SiteInteractive {
    param([Parameter(Mandatory)] $Sites)

    while ($true) {
        $q = Read-Host "Site (partial name or ID; '?' to list all; blank to cancel)"
        if ([string]::IsNullOrWhiteSpace($q)) { return $null }
        if ($q -eq '?') { $Sites | Format-Table Id, Name -AutoSize | Out-Host; continue }

        if ($q -match '^\d+$') {
            $exact = $Sites | Where-Object { $_.Id -eq $q }
            if ($exact) { return $exact }
        }

        $m = @($Sites | Where-Object {
            $_.Name -match [regex]::Escape($q) -or $_.Id -match [regex]::Escape($q)
        })
        if ($m.Count -eq 0) { Write-Host "  no matches" -ForegroundColor Yellow; continue }
        if ($m.Count -eq 1) {
            Write-Host "  $($m[0].Id)  $($m[0].Name)" -ForegroundColor Cyan
            return $m[0]
        }

        Write-Host "  $($m.Count) matches:"
        for ($i = 0; $i -lt $m.Count; $i++) {
            Write-Host "  [$($i+1)] $($m[$i].Id)  $($m[$i].Name)"
        }
        $pick = Read-Host "Pick # (blank to search again)"
        if ($pick -match '^\d+$' -and [int]$pick -ge 1 -and [int]$pick -le $m.Count) {
            return $m[[int]$pick - 1]
        }
    }
}


# ---------------------------------------------------------------------
# Select-DateRange
#   Presets + custom range. Returns @{From;To} or $null on cancel.
#   No 7-day cap — the UI limit applies to the viewer only.
# ---------------------------------------------------------------------
function Select-DateRange {
    $today = (Get-Date).Date
    Write-Host ""
    Write-Host "  [1] Yesterday only (default)"
    Write-Host "  [2] Today"
    Write-Host "  [3] Last 7 days"
    Write-Host "  [4] Last 30 days"
    Write-Host "  [5] Custom range"
    Write-Host "  [0] Cancel"
    switch (Read-Host "Select") {
        '1' { return @{ From = $today.AddDays(-1); To = $today.AddDays(-1) } }
        '2' { return @{ From = $today;              To = $today } }
        '3' { return @{ From = $today.AddDays(-6);  To = $today } }
        '4' { return @{ From = $today.AddDays(-29); To = $today } }
        '5' {
            while ($true) {
                $f = Read-Host "  From yyyy-MM-dd [$($today.AddDays(-1).ToString('yyyy-MM-dd'))]"
                if ([string]::IsNullOrWhiteSpace($f)) { $f = $today.AddDays(-1).ToString('yyyy-MM-dd') }
                $fd = [datetime]::MinValue
                if (-not [datetime]::TryParse($f, [ref]$fd)) { Write-Host "  bad date" -ForegroundColor Yellow; continue }

                $t = Read-Host "  To   yyyy-MM-dd [$($fd.ToString('yyyy-MM-dd'))]"
                if ([string]::IsNullOrWhiteSpace($t)) { $t = $fd.ToString('yyyy-MM-dd') }
                $td = [datetime]::MinValue
                if (-not [datetime]::TryParse($t, [ref]$td)) { Write-Host "  bad date" -ForegroundColor Yellow; continue }
                if ($td.Date -lt $fd.Date) { Write-Host "  'To' is before 'From'" -ForegroundColor Yellow; continue }

                return @{ From = $fd.Date; To = $td.Date }
            }
        }
        default { return $null }
    }
}


# ---------------------------------------------------------------------
# Invoke-WebEorDownloaderMenu
#   Interactive driver: site / date range / output path / download.
# ---------------------------------------------------------------------
function Invoke-WebEorDownloaderMenu {

    $sites       = $null
    $selSiteId   = $null
    $selSiteName = $null
    $fromDate    = $null
    $toDate      = $null
    $outFile     = $null
    $conn        = $null

    while ($true) {
        Clear-Host
        $siteLine = if ($selSiteId) { "$selSiteName ($selSiteId)" } else { "(not selected)" }
        $dateLine = if ($fromDate) {
            if ($fromDate -eq $toDate) {
                $fromDate.ToString('yyyy-MM-dd')
            } else {
                "$($fromDate.ToString('yyyy-MM-dd'))  to  $($toDate.ToString('yyyy-MM-dd'))  ($((($toDate - $fromDate).Days + 1)) days)"
            }
        } else { "(not selected)" }
        $outLine  = if ($outFile) { $outFile } else { "(auto-generated)" }

        Write-Host "=============================================================="
        Write-Host "  WebEOR  EOR CSV Downloader"
        Write-Host "=============================================================="
        Write-Host ""
        Write-Host "   1. Site         : $siteLine"
        Write-Host "   2. Date range   : $dateLine"
        Write-Host "   3. Output file  : $outLine"
        Write-Host ""
        Write-Host "   4. Download CSV"
        Write-Host "   Q. Quit"
        Write-Host ""

        switch ((Read-Host "  Select").ToUpper()) {

            '1' {
                if (-not $sites) {
                    Write-Host "  Loading site list..."
                    $sites = Get-SiteList
                }
                $pick = Select-SiteInteractive -Sites $sites
                if ($pick) {
                    $selSiteId   = $pick.Id
                    $selSiteName = $pick.Name
                    $conn        = $null
                    $outFile     = $null
                }
            }

            '2' {
                $r = Select-DateRange
                if ($r) { $fromDate = $r.From; $toDate = $r.To; $outFile = $null }
            }

            '3' {
                $default = if ($selSiteId -and $fromDate) {
                    "EOR_${selSiteId}_$($fromDate.ToString('yyyyMMdd'))_$($toDate.ToString('yyyyMMdd')).csv"
                } else { "EOR.csv" }
                $defaultFull = Join-Path (Get-Location) $default
                $typed = Read-Host "  Path [$defaultFull]"
                $outFile = if ([string]::IsNullOrWhiteSpace($typed)) { $defaultFull } else { $typed }
            }

            '4' {
                if (-not $selSiteId) { Write-Host "  Pick a site first." -ForegroundColor Yellow; Start-Sleep 1; continue }
                if (-not $fromDate)  { Write-Host "  Pick a date range first." -ForegroundColor Yellow; Start-Sleep 1; continue }

                if (-not $outFile) {
                    $outFile = Join-Path (Get-Location) `
                        "EOR_${selSiteId}_$($fromDate.ToString('yyyyMMdd'))_$($toDate.ToString('yyyyMMdd')).csv"
                }

                $dayCount = ($toDate - $fromDate).Days + 1
                if ($dayCount -gt 7) {
                    Write-Host "  Note: $dayCount days selected. The UI limits the *viewer* to 7 days," -ForegroundColor Yellow
                    Write-Host "        but the CSV endpoint accepts any range. Continuing." -ForegroundColor Yellow
                }

                try {
                    if (-not $conn -or $conn.SiteId -ne [int]$selSiteId) {
                        Write-Host "  Logging in as site $selSiteId..." -ForegroundColor Cyan
                        $conn = Connect-WebEor -SiteId ([int]$selSiteId)
                    }

                    Write-Host "  Downloading..." -ForegroundColor Cyan
                    $csv = Get-WebEorCsv -Connection $conn -SiteId ([int]$selSiteId) `
                        -FromDate $fromDate -ToDate $toDate -OutFile $outFile

                    $lines = (Get-Content $csv.FullName | Measure-Object -Line).Lines
                    Write-Host ""
                    Write-Host "  Saved: $($csv.FullName)" -ForegroundColor Green
                    Write-Host "         $([math]::Round($csv.Length/1KB,1)) KB,  ~$($lines - 1) data rows" -ForegroundColor Green
                    Write-Host ""
                    Write-Host "  Preview (first 3 rows):" -ForegroundColor DarkGray
                    Get-Content $csv.FullName -TotalCount 4 | ForEach-Object {
                        Write-Host "    $_" -ForegroundColor DarkGray
                    }
                } catch {
                    Write-Host ""
                    Write-Host "  FAILED: $($_.Exception.Message)" -ForegroundColor Red
                }
                Read-Host "  Press Enter"
            }

            'Q' { return }
        }
    }
}
