$session = New-Object Microsoft.PowerShell.Commands.WebRequestSession

# 1. Establish session (cookie, Windows auth handshake, whatever the app needs)
Invoke-WebRequest -Uri "https://webeor.usps.gov/EORViewer/GUI_EORView_Viewer.asp" `
                  -WebSession $session -UseDefaultCredentials | Out-Null

# 2. The SQL — paste verbatim from the form, keep the line breaks, keep <>
$sql = @'
SELECT /*+ ordered */ SUBSTR( LPAD( postal_facility.zip_code_id, 9, '0' ), 1, 5 ) || '-' || SUBSTR( LPAD( postal_facility.zip_code_id, 9, '0' ), 6 ) site_code, machine_type.mach_type_desc, machine.machine_nbr, TRIM(application_data.machine_sort_program_name) msp, application_data.tour_code, TO_CHAR( application_data.start_dtm, 'MM/DD/YY HH24:MI') start_date, TO_CHAR( application_data.end_dtm, 'MM/DD/YY HH24:MI') end_date, GUI_Common_Helper.FormatOpNbr( application_data.mail_operation_nbr ) op_nbr, application_data.app_sequence_nbr, application_data.total_pieces_fed_cnt, GUI_Common_Helper.GetRunTransStatus( application_data.mail_operation_nbr, application_data.mods_trans_ind, application_data.mods_trans_date ) ModsTrans, DECODE( UPPER( application_data.dsis_ind ), 'T', 'S', 'NS' ) dsis, end_of_run.run_nbr, application_data.run_sequence_nbr , substr(TO_CHAR(application_data.mods_date, 'MM/DD/YY HH24:MI'), 7, 2) mYear, substr(TO_CHAR(application_data.start_dtm, 'MM/DD/YY HH24:MI'), 7, 2) sYear, substr(TO_CHAR(application_data.end_dtm, 'MM/DD/YY HH24:MI'), 7, 2) eYear, NVL(substr(TO_CHAR(application_data.mods_trans_date), 9, 2), '00') mTransYear, postal_facility.site_name FROM EOR_PSBRPA.application_data, EOR_PSBRPA.end_of_run, ( SELECT site_id, site_name, zip_code_id FROM EOR_PSBRPA.postal_facility x START WITH x.site_id = 19010 CONNECT BY PRIOR x.site_id = x.parent_site_id AND x.site_id <> 19010 ) postal_facility, EOR_PSBRPA.machine_type, EOR_PSBRPA.machine WHERE end_of_run.site_id = postal_facility.site_id AND end_of_run.machine_id = machine.machine_id AND end_of_run.site_id = machine.site_id AND end_of_run.run_mach_type_code = machine_type.mach_type_code AND end_of_run.run_sequence_nbr = application_data.run_sequence_nbr AND application_data.mods_date >= TO_DATE('17-Oct-2026', 'DD-MON-YYYY') AND application_data.mods_date <= TO_DATE('17-Oct-2026', 'DD-MON-YYYY') ORDER BY mach_type_desc asc, machine_nbr asc,  myear asc, syear asc, start_date asc
'@

$body = @{
    tabCsvGroup   = " "
    tabCsvHeader  = "Site,MType,MNo,Op No.,Sort Program,Tour,Run#,Start,End,Fed,MODS,DOIS"
    tabCsvColumns = "0,1,2,7,3,4,12,5,6,9,10,11"
    tabCsvSource  = "CLIENT"
    tabCsvSQL     = $sql
}

$resp = Invoke-WebRequest -Uri "https://webeor.usps.gov/ReportPage/GUI_Report_CSV.asp" `
                          -Method POST -Body $body -WebSession $session `
                          -UseDefaultCredentials -ErrorAction Stop

$resp.Content | Out-File "$env:TEMP\test_eor.csv" -Encoding UTF8
"HTTP $($resp.StatusCode)  |  $($resp.RawContentLength) bytes  |  CT=$($resp.Headers['Content-Type'])"
